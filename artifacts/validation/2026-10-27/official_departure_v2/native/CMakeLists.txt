cmake_minimum_required(VERSION 3.16)
project(arena_material_trial LANGUAGES CXX)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(material_client material_client.cpp)
target_compile_features(material_client PRIVATE cxx_std_17)
target_compile_options(material_client PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(material_client PRIVATE gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
