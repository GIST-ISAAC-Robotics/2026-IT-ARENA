def test_policy_default_and_explicit_measured_selection():
    from arena_autonomy.parallel_lane_trial import ParallelLaneTrial
    from arena_autonomy.rear_lane_keep_trial import RearLaneKeepSelector
    import arena_autonomy.observed_lane_preference_trial as v3
    from arena_autonomy.measured_parallel_trial import MeasuredParallelTrial
    assert FollowConfig().rear_lane_keep_policy == 'timed' and FollowConfig().rear_lane_keep_trial is False
    timed = LocalFollow(config(rear_lane_keep_policy='timed'))
    assert type(timed.pass_selector) is RearLaneKeepSelector and type(timed.parallel_trial) is ParallelLaneTrial
    observed = LocalFollow(config(rear_lane_keep_policy='observed'))
    # 기존 observed(v3)의 의미와 평행 후보는 그대로다.
    assert type(observed.pass_selector) is v3.StraightLanePreferenceSelector
    assert type(observed.parallel_trial) is ParallelLaneTrial
    measured = LocalFollow(config())
    assert type(measured.pass_selector) is lk.StraightLanePreferenceSelector
    assert lk.StraightLanePreferenceSelector is not v3.StraightLanePreferenceSelector
    assert type(measured.parallel_trial) is MeasuredParallelTrial
    for value in ('observed_measured', 'observed_measured_v2', 'v4', '', None, True, 1):
        with pytest.raises(ValueError):
            config(rear_lane_keep_policy=value)
    with pytest.raises(ValueError):
        FollowConfig(rear_lane_keep_policy='observed_measured_v1')
    with pytest.raises(ValueError):
        FollowConfig(**{**CHAIN, 'rear_lane_keep_policy': 'observed_measured_v1'})


def test_runtime_copy_matches_offline_v4_except_paths_description_and_cap_boundary():
    from pathlib import Path
    import arena_autonomy.measured_parallel_trial as state
    root = Path(__file__).resolve().parents[1]
    previous = root/V4

    def body(source):
        return source[source.index('import math'):]
    assert body(Path(lk.__file__).read_text(encoding='utf-8')).replace(
        'from arena_autonomy.measured_parallel_speed_trial import', 'from parallel_speed import') == body(
        (previous/'candidate.py').read_text(encoding='utf-8'))
    assert body(Path(state.__file__).read_text(encoding='utf-8')).replace(
        'from arena_autonomy.measured_parallel_speed_trial import', 'from parallel_speed import') == body(
        (previous/'parallel_state.py').read_text(encoding='utf-8'))
    fixed = body(Path(parallel_speed.__file__).read_text(encoding='utf-8'))
    original = body((previous/'parallel_speed.py').read_text(encoding='utf-8'))
    for before, after in [
            ('import math\n', 'import math\nimport numbers\n'),
            ("    if (isinstance(target_cap, bool) or not math.isfinite(target_cap)\n"
             "            or not 0 < target_cap <= config.max_speed_mps):\n"
             "        return {**result, 'reason': 'invalid_target_cap'}\n",
             "    if (isinstance(target_cap, bool) or not isinstance(target_cap, numbers.Real)\n"
             "            or not math.isfinite(target_cap) or not target_cap > 0):\n"
             "        return {**result, 'reason': 'invalid_target_cap'}\n"
             "    # 명령 상한은 설정 최대속도와 기존 경로 cap의 min이다. 측정속도 유효성 검사와 분리한다.\n"
             "    target_cap = min(float(target_cap), config.max_speed_mps)\n")]:
        assert original.count(before) == 1
        original = original.replace(before, after)
    assert fixed == original


@pytest.mark.parametrize('maximum', [.2, .3, .34999])
def test_default_cap_below_configured_maximum_is_not_rejected(measured_case, maximum):
    """v4 원본은 이 설정에서 invalid_target_cap으로 평행 후보를 항상 HOLD했다(10/30 반례)."""
    from unittest.mock import patch
    import arena_autonomy.parallel_lane_trial as original
    scene, base, state = measured_case
    cfg = config(max_speed_mps=maximum)
    result = parallel_speed.proposal(scene, base, .9*maximum, 0., .02, cfg)
    assert result['eligible'] and result['reason'] == 'parallel_current_position'
    assert 0 < result['target_speed_mps'] <= cfg.max_speed_mps
    with patch.object(original, 'per_ray_padding', parallel_speed.per_ray_padding), \
            patch.object(original, 'swept_limit', parallel_speed.swept_limit):
        assert result == original.proposal(scene, base, .9*maximum, 0., .02, cfg)
    assert not state['captured']


@pytest.mark.parametrize('maximum', [.2, .3, 1.])
@pytest.mark.parametrize('cap', [.35, .5, 1., 5.])
def test_command_never_exceeds_configuration_for_any_requested_cap(measured_case, maximum, cap):
    scene, base, state = measured_case
    cfg = config(max_speed_mps=maximum)
    result = parallel_speed.proposal(scene, base, .5*maximum, 0., .02, cfg, cap)
    assert result['eligible']
    assert result['target_speed_mps'] <= min(cap, cfg.max_speed_mps)+1e-12

@pytest.mark.parametrize('speed', [.31, .5])
def test_measured_above_low_maximum_needs_own_region_and_command_stays_capped(measured_case, speed):
    scene, base, state = measured_case
    cfg = config(max_speed_mps=.3)
    result = parallel_speed.proposal(scene, base, speed, 0., .02, cfg)
    assert result['eligible'] and result['measured_above_target_cap'] is True
    assert result['measured_speed_mps'] == speed
    assert result['target_speed_mps'] <= cfg.max_speed_mps
    assert state['captured'] == [(speed, result['target_speed_mps'])]
    state['unobserved'] = 1
    blocked = parallel_speed.proposal(scene, base, speed, 0., .02, cfg)
    assert not blocked['eligible'] and blocked['reason'] == 'measured_region_unobserved'
    state['unobserved'], state['clearance'] = 0, .2
    short = parallel_speed.proposal(scene, base, speed, 0., .02, cfg)
    assert not short['eligible'] and short['reason'] in ('measured_stop_space_insufficient', 'unchanged_gap_hold')

def test_low_maximum_parallel_trial_moves_after_fresh_confirmations(measured_case):
    from arena_autonomy.measured_parallel_trial import MeasuredParallelTrial
    scene, base, state = measured_case
    cfg = config(max_speed_mps=.3)
    trial = MeasuredParallelTrial(cfg)
    first = trial.apply(scene, base, .2, 0., .02)
    assert first[0] == 0 and first[2]['reason'] == 'parallel_hold_release_wait'
    # 같은 스캔의 반복 호출은 재개 확인 수를 늘리지 않는다.
    assert trial.apply(scene, base, .2, 0., .02)[0] == 0
    scene.source_first_s, scene.t_ref_s = .1, .2
    second = trial.apply(scene, base, .2, 0., .02)
    assert 0 < second[0] <= cfg.max_speed_mps


def test_low_maximum_selector_request_cap_stays_below_configuration(setup, monkeypatch):
    _, state, _ = setup
    ctl = lk.StraightLanePreferenceSelector(config(max_speed_mps=.3))

    def call(t, *, speed=.2, base=None, par=None):
        scene = SimpleNamespace(source_first_s=t-.1, t_ref_s=t, source_epoch=0,
                                occupied_endpoints_xy_m=state['points'], rays=[SimpleNamespace(acquired_at_s=t-.0002)])
        return ctl.select(scene, baseline(.3) if base is None else base, parallel(.3) if par is None else par,
                          speed, 0., .02)
    release_from_rear(state, call)
    assert ctl.mode == 'RELEASED' and 'left' in ctl.lane_keep
    caps = []

    def proposal(scene, base, speed, previous, dt, cfg, cap):
        caps.append(cap)
        return dict(eligible=True, reason='parallel_current_position', target_speed_mps=cap,
                    steering_rad=0., path_points=PATH, clearance_m=1.5, maximum_padding_m=.1)
    monkeypatch.setattr(parallel_speed, 'proposal', proposal)
    out = call(1.3, base=baseline(.3, .2))
    assert caps == pytest.approx([.3-lk.ENCODER_SPEED_QUANTUM_MPS])
    assert out[2]['rear_lane_keep']['applied'] is True
    assert 0 < out[0] <= .3
