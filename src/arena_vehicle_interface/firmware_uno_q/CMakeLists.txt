cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(arena_uno_q LANGUAGES C CXX)
if(NOT ARENA_PROFILE_HEADER OR NOT ST_DRIVER_DIR)
  message(FATAL_ERROR "Explicit generated profile and pinned ST driver are required")
endif()
target_sources(app PRIVATE src/main.cpp src/hal.cpp src/imu.cpp "${ST_DRIVER_DIR}/lsm6dsox_reg.c")
target_include_directories(app PRIVATE include ../portable ../firmware_runtime_preview/main "${ST_DRIVER_DIR}")
target_compile_options(app PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=c++17>)
target_compile_definitions(app PRIVATE ARENA_PROFILE_HEADER="${ARENA_PROFILE_HEADER}")
